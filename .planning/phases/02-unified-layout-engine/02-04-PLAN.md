**generatePDF.js — Zielzustand:**

Die `generatePDF(format)` Funktion wird zu:
```javascript
// Neue Imports am Anfang hinzufügen (nach bestehenden Imports)
import { calculateLayout } from './layout/engine.js';
import { renderToPDF } from './layout/pdfRenderer.js';
import { loadFontArrayBuffers } from './layout/fontManager.js';

async function generatePDF(format) {
    const progressContainer = document.getElementById('pdf-progress-container');
    progressContainer.style.display = 'block';
    showProgress(0, "Initialisiere PDF-Erstellung");

    // Config lesen (aus localStorage wie bisher)
    let config;
    try {
        const saved = localStorage.getItem('liedblattConfig');
        config = saved ? JSON.parse(saved) : {};
    } catch (e) {
        config = {};
    }
    config.format = format || config.format || 'a5';

    // fontSize: script.js speichert px-Werte, Engine erwartet pt
    const fontSizePt = (parseFloat(config.fontSize) || 12) * 0.75;

    const engineConfig = {
        format: config.format,
        fontSize: fontSizePt,
        lineHeight: parseFloat(config.lineHeight) || 1.5,
        textAlign: config.textAlign || 'left',
        fontFamily: config.fontFamily || 'Jost',
    };

    showProgress(10, "Lade Schriftarten");
    const arrayBuffers = await loadFontArrayBuffers(engineConfig.fontFamily);

    // Temporäres PDFDocument für font.widthOfTextAtSize (Engine braucht PDFFont-Objekte)
    const { PDFDocument } = window.PDFLib;
    const fontkit = window.fontkit;
    const tempDoc = await PDFDocument.create();
    tempDoc.registerFontkit(fontkit);
    const { embedFontsInDoc } = await import('./layout/fontManager.js');
    const fonts = await embedFontsInDoc(tempDoc, arrayBuffers);

    showProgress(30, "Berechne Layout");
    const items = Array.from(document.getElementById('liedblatt-content').children);
    const layoutResult = await calculateLayout(items, engineConfig, fonts);

    showProgress(60, "Rendere PDF");
    // Logo-ArrayBuffer laden falls vorhanden
    let logoArrayBuffer = null;
    if (config.churchLogo) {
        try {
            const token = localStorage.getItem('token');
            const logoRes = await fetch(config.churchLogo, {
                headers: { 'Authorization': `Bearer ${token}` }
            });
            if (logoRes.ok) logoArrayBuffer = await logoRes.arrayBuffer();
        } catch (e) {
            console.warn('generatePDF: Logo nicht ladbar:', e.message);
        }
    }

    const pdfDoc = await renderToPDF(layoutResult, engineConfig, arrayBuffers, logoArrayBuffer);

    showProgress(90, "Finalisiere PDF");
    const pdfBytes = await pdfDoc.save();

    // Broschüren-Option bleibt erhalten — createBrochure() existiert noch
    const brochureCheckbox = document.getElementById('brochure-checkbox');
    const makeBrochure = brochureCheckbox && brochureCheckbox.checked;
    const finalBytes = makeBrochure ? await createBrochure(pdfBytes, config.format) : pdfBytes;

    downloadPDF(finalBytes, `liedblatt-${config.format}.pdf`);
    showProgress(100, "Fertig");
    setTimeout(() => { progressContainer.style.display = 'none'; }, 2000);
}
```

Alles weitere in generatePDF.js (fetchAndEmbedFont, splitTextToLines, alle drawXxx-Hilfsfunktionen, estimatePDFElementHeight, processElementGroups usw.) kann stehen bleiben — es schadet nicht, solange es nicht aufgerufen wird. Der wichtige Punkt ist: die neue `generatePDF()` Funktion ruft nur noch calculateLayout + renderToPDF auf.

Entferne das lokale `const globalConfig = { ... }` aus der generatePDF-Funktion (Zeile 881-888) — es wird nicht mehr benötigt.

WICHTIG: Das `window.generatePDF = generatePDF;` Statement am Anfang (Zeile 5) bleibt erhalten.

**previewPageBreaks.js — Zielzustand:**

Die `updatePreviewWithPageBreaks(format)` Funktion wird zu:
```javascript
// Neue Imports am Anfang der Datei hinzufügen
import { calculateLayout } from './layout/engine.js';
import { renderToDOM } from './layout/domRenderer.js';
import { loadFontArrayBuffers, embedFontsInDoc } from './layout/fontManager.js';

// Debounce-Variable von 300ms auf 150ms ändern:
// calculateTimeout = setTimeout(() => { ... }, 300)  →  150

export async function updatePreviewWithPageBreaks(format = 'a5') {
    if (calculateTimeout) clearTimeout(calculateTimeout);

    calculateTimeout = setTimeout(async () => {
        if (isCalculating) return;
        isCalculating = true;

        try {
            const container = document.getElementById('liedblatt-content');
            if (!container) return;

            // Config aus globalConfig oder localStorage
            let config;
            try {
                const saved = localStorage.getItem('liedblattConfig');
                config = saved ? JSON.parse(saved) : {};
            } catch (e) { config = {}; }

            config.format = format;
            const fontSizePt = (parseFloat(config.fontSize) || 12) * 0.75;
            const engineConfig = {
                format,
                fontSize: fontSizePt,
                lineHeight: parseFloat(config.lineHeight) || 1.5,
                textAlign: config.textAlign || 'left',
                fontFamily: config.fontFamily || 'Jost',
            };

            // Font-Objekte für Engine (calculateLayout braucht PDFFont.widthOfTextAtSize)
            const arrayBuffers = await loadFontArrayBuffers(engineConfig.fontFamily);
            const { PDFDocument } = window.PDFLib;
            const fontkit = window.fontkit;
            const tempDoc = await PDFDocument.create();
            tempDoc.registerFontkit(fontkit);
            const fonts = await embedFontsInDoc(tempDoc, arrayBuffers);

            const items = Array.from(container.children);
            const layoutResult = await calculateLayout(items, engineConfig, fonts);

            // DOM-Renderer befüllt container komplett neu
            renderToDOM(layoutResult, engineConfig, container);

            // ENTFERNT: window.lastCalculatedBreakPositions = breakPositions (war Zeile 76)

        } catch (error) {
            console.error("Fehler bei der Vorschau-Aktualisierung:", error);
        } finally {
            isCalculating = false;
        }
    }, 150); // war 300ms — per Entscheidung auf 150ms
}
```

`initPreviewFormatSelector()` bleibt strukturell erhalten, ruft aber das neue `updatePreviewWithPageBreaks()` auf. Prüfe ob `export default { ... }` am Ende der Datei (Zeile 1005) mit den named exports kollidiert — falls ja, entferne den default-Export (named exports sind ausreichend und werden schon aus script.js importiert).
