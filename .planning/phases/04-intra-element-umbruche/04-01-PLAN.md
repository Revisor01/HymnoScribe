async function pushTextBlock(text, fontSize, fontStyle, alignment, extraSpacingAfter, flags) {
    const isBold   = fontStyle === 'bold'   || fontStyle === 'boldItalic';
    const isItalic = fontStyle === 'italic'  || fontStyle === 'boldItalic';
    const font = getFontForStyle(fonts, isBold, isItalic);
    const lines = await splitTextToLines(text, font, fontSize, contentWidth);
    const lineHeightPt = fontSize * lineHeightFactor;
    const blockHeight = lines.length * lineHeightPt;

    // Seitenumbruch falls kein Platz — HIER muss Split-Logik rein
    if (currentY + blockHeight > pageSize.height - MARGINS.bottom) {
        newPage(); // aktuell: ganzer Block naechste Seite
    }

    currentBlocks.push({
        type: 'text',
        x: MARGINS.left,
        y: currentY,
        width: contentWidth,
        height: blockHeight,
        data: {
            lines,
            fontSize,
            fontStyle,
            alignment: alignment || config.textAlign || 'left',
            lineHeight: lineHeightFactor,
            ...flags
        }
    });
    currentY += blockHeight + (extraSpacingAfter || 0);
}

  <action>
`pushTextBlock()` (Zeilen 193–223) so umschreiben, dass es Bloecke zeilenweise splittet. Der Import von `LAYOUT` aus constants.js muss oben in der Datei vorhanden sein — pruefen und ggf. erganzen.

**Logik (iterativer Loop statt Rekursion — kein Stack-Overflow-Risiko):**

1. `lines[]` und `lineHeightPt` wie bisher berechnen
2. Statt einfachem `if (blockHeight > verfuegbar) newPage()` — While-Loop:
   ```
   remaining = lines[]; // alle Zeilen noch zu platzieren
   isFirstChunk = true;
   
   while remaining nicht leer:
     available = (pageSize.height - MARGINS.bottom) - currentY
     fitsCount = Math.floor(available / lineHeightPt)
     
     if fitsCount < MIN_LINES_BEFORE_SPLIT:
       newPage()
       continue  // naechste While-Iteration berechnet fitsCount neu
     
     if fitsCount >= remaining.length:
       // Alles passt auf diese Seite — Block fertig platzieren
       chunk = remaining
       remaining = []
       spacingAfter = extraSpacingAfter
     else:
       // Split: erste fitsCount Zeilen hierher, Rest naechste Seite
       chunk = remaining.slice(0, fitsCount)
       remaining = remaining.slice(fitsCount)
       spacingAfter = 0  // kein Spacing nach erstem Teil (Seite voll)
     
     currentBlocks.push({
       type: 'text',
       x: MARGINS.left,
       y: currentY,
       width: contentWidth,
       height: chunk.length * lineHeightPt,
       data: {
         lines: chunk,
         fontSize,
         fontStyle,
         alignment: alignment || config.textAlign || 'left',
         lineHeight: lineHeightFactor,
         isSplitContinuation: !isFirstChunk,
         ...flags
       }
     })
     currentY += chunk.length * lineHeightPt + spacingAfter
     isFirstChunk = false
     
     if remaining nicht leer: newPage()
   ```

**Kritische Details:**
- `isSplitContinuation: !isFirstChunk` in `data` — kostenlos, ermoeglicht spaetere visuelle Erweiterungen
- Erster Teil bekommt `spacingAfter = 0` (Seite voll, kein Platz fuer Abstand)
- Letzter Teil bekommt `spacingAfter = extraSpacingAfter` (originaler Abstand nach dem Element)
- `LAYOUT` muss aus constants.js importiert sein — pruefen: `import { ..., LAYOUT } from './constants.js'`
- Funktionsname bleibt `pushTextBlock()` — alle Aufrufer (Zeilen ~393–399) bleiben unveraendert
